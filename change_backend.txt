    offset: number;
    limit: number;
  }

  type PaginationMeta = PaginationMetaWithTotal | PaginationMetaWithHasMore;

  // Helper Function
  function getResultSummary(meta: PaginationMeta, itemCount: number): string {
    if ('total' in meta) {
      return `พบ ${meta.total} รายการ`;
    } else if ('hasMore' in meta) {
      return meta.hasMore
        ? `พบ ${itemCount}+ รายการ`
        : `พบ ${itemCount} รายการ`;
    }
    return '';
  }

  ⚠️ Breaking Change หรือไม่?

  ใช่ - เป็น Breaking Change เพราะ:

  | ปัญหา                                             | ผลกระทบ                          |
  |---------------------------------------------------|----------------------------------|
  | meta.total เปลี่ยนจาก number → number | undefined | TypeScript จะ error ถ้าใช้โดยตรง |
  | Search endpoints ไม่ส่ง total แล้ว                | UI ที่แสดง "พบ X รายการ" จะพัง   |
  | Pagination logic ที่ใช้ total คำนวณหน้า           | ต้องเปลี่ยนเป็นใช้ hasMore       |

  ---
  💡 ถ้าไม่อยากให้ Breaking ต้องทำยังไง?

  ทางเลือก A: Backend ส่งทั้ง total และ hasMore

  ให้ผมแก้ Backend ส่งทั้งคู่:

  // Search endpoints จะส่ง
  {
    "meta": {
      "total": 20,      // ← ค่าประมาณ (= posts.length)
      "hasMore": true,  // ← ค่าจริง
      "offset": 0,
      "limit": 20
    }
  }

  ข้อดี: Frontend ไม่พัง
  ข้อเสีย: total เป็นค่าหลอก (ไม่ตรงกับความจริง)

  ทางเลือก B: API Versioning

  สร้าง /api/v2/posts/search ใหม่ที่ใช้ hasMore
  ให้ /api/v1/posts/search ยังส่ง total แบบเดิม

  ข้อดี: ไม่ Breaking
  ข้อเสีย: Maintain 2 versions

  ทางเลือก C: Gradual Migration (แนะนำ)

  1. Phase 1: Backend ส่งทั้ง total และ hasMore พร้อมกัน
  2. Phase 2: Frontend แก้โค้ดใช้ hasMore
  3. Phase 3: Backend เลิกส่ง total ใน search endpoints

  - ✅ ถูกต้อง ชัดเจน ไม่หลอกลวง

  - ❌ Frontend ต้องแก้

  - ✅ ไม่ Breaking
  - ❌ total เป็นค่าประมาณ (หลอก ๆ)
